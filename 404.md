<script>
    document.addEventListener("DOMContentLoaded", function() {
        // Controlla se l'URL attuale o quello di provenienza indica la lingua italiana
        const isItalian = window.location.pathname.startsWith('/it/');
        
        if (isItalian) {
            // Aggiorna il titolo della pagina nel browser
            document.title = "404: Pagina non trovata";
            
            // Aggiorna il messaggio principale
            const message = document.getElementById('404-message');
            message.innerText = "Oops! La pagina che stai cercando non esiste o è stata spostata.";
            
            // Aggiorna il bottone "Torna alla Home"
            const link = document.getElementById('404-link');
            link.innerText = "{{ site.data.it.back_to_home }}";
            link.href = "{{ '/it/' | relative_url }}";
        }
    });
</script>
